## Bin resolution for each dimension
fsc_small_bin_count <- 80
pe_bin_count <- 40
chl_small_bin_count <- 40

## Create grid bins.
grid_bins <- list()
grid_bins$fsc_small <- 2^seq(from = log2(0.0003), to = log2(3200),
                             length = fsc_small_bin_count+1)
grid_bins$pe <- 2^seq(from = log2(1),
                      to = log2(3200),
                      length = pe_bin_count+1)
grid_bins$chl_small <- 2^seq(from = log2(1),
                             to = log2(3200),
                             length = chl_small_bin_count+1)
