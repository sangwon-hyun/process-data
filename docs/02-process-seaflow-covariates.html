## Helper function for plotting matrix heatmaps
drawmat_precise_ggplot <- function(distmat, colours = c("blue", "red"),
                                   limits = NULL,
                                   xlab = "", ylab = "", title = "") {
  distmat <- as.matrix(distmat)
  longData <- reshape2::melt(distmat[nrow(distmat):1, ])

  p <- ggplot() +
    geom_raster(aes(x = Var2, y = Var1, fill = value), data = longData) +
    scale_fill_gradientn(colours = colours, guide = "colorbar", limits = limits) +
    labs(x = xlab, y = ylab, title = title) +
    theme_minimal() + 
    theme(
      axis.text.x = element_text(size = rel(1), angle = 90, vjust = 0.3),
      axis.text.y = element_text(size = rel(1)),
      plot.title = element_text(size = rel(1.5))
    )
  return(p)
}

make_html_table <- function(mytab, table_caption = NULL) {
  mytab |>
    kable(
      format = "html", 
      digits = 2, 
      align = "c",
      caption = table_caption
    ) |>
    kable_styling(
      bootstrap_options = c("striped", "hover", "condensed"),
      full_width = FALSE
    )
}

for(cruisename in cruise_ids){
  cat(cruisename, "\n")

  figure_dir <- file.path("figures", cruisename)
  if(!dir.exists(figure_dir)) dir.create(figure_dir)

  ## Load data
  X <- read.csv(file = file.path(datadir, cruisename, "X.csv")) %>% as_tibble()
  
  ## Extract spatiotemporal identifiers and scale numerical variables
  time0 <- lubridate::as_datetime(X$time)
  lat0 <- X %>% pull(lat)
  lon0 <- X %>% pull(lon)
  varnames <- X %>% select(-c("time", "lat", "lon")) %>% colnames()
  
  ## 1. First take the means and standard deviations
  means = X %>% select(-time, -lat, -lon) %>%
    summarise(across(everything(), ~mean(.x, na.rm = TRUE))) 
  sds = X %>% select(-time, -lat, -lon) %>%
    summarise(across(everything(), ~sd(.x, na.rm = TRUE))) 
  
  ## Base R plot
  means_sorted <- apply(X[-c(1,2,3)], 2, mean, na.rm=TRUE) %>%
    sort(decreasing=TRUE)

  png(
    file.path(figure_dir, "variable_means.png"),
    width = 8, height = 5, units = "in", res = 150
  )
    plot(means_sorted, main = "Means of all variables, sorted")
  graphics.off()
  
  ## 2. Then, identify which variables have really big means
  thresh = 1E10
  somenames = means[,which(means>thresh & sds > thresh)] %>% names()
  
  if(length(somenames) >= 1){
    ## 3. Visualize 
    p <- X[,somenames] %>%
      add_column(time = time0) %>%
      pivot_longer(-time)  %>% 
      ggplot() +
      geom_point(aes(x=time, y = value, group = name), size = .5) +
      facet_wrap(~name, scale = 'free_y')   +
      ggtitle("The unusually high-value variables")
    
    ggsave(
      file.path(figure_dir, "high_value_variables.png"),
      p, width = 8, height = 5, units = "in"
    )
    
    ## 4. Remove these variables
    X = X %>% select(-all_of(somenames))
  }
  
  ## Store the scaling information
  scaleobj <- X %>% select(-lat, -lon, -time) %>% scale() 
  mean_val <- attr(scaleobj, "scaled:center") %>% unname()
  sd_val <- attr(scaleobj, "scaled:scale") %>% unname()
  
  ## Double check that scaling is sound
  scaled_data = as.matrix(scaleobj)
  unscaled_data <- sweep(scaled_data, 2, sd_val, `*`)
  unscaled_data <- sweep(unscaled_data, 2, mean_val, `+`)
  orig_matrix <- as.matrix(X %>% select(-time, -lat, -lon))
  
  ## Base R plot
  max_diffs <- apply(abs(unscaled_data - orig_matrix), 2, max, na.rm = TRUE) %>%
    sort()

  png(
    file.path(figure_dir, "reconstruction_errors.png"),
    width = 8, height = 5, units = "in", res = 150
  )
    plot(max_diffs, type = "p", xlab = "Variables (Sorted)",
        ylab = "Max Absolute Difference",
        main = "Reconstruction Error per Variable")
    abline(h = 0, col = "red", lty = 2)
  graphics.off()
  
  ## Save the scaling information
  X_scaled = scaled_data

  if(!dir.exists(file.path(outputdir, cruisename))) dir.create(file.path(outputdir, cruisename))
  
  saveRDS(list(X_before_scaling = X,
               X_after_scaling = X_scaled,
               scaleobj = scaleobj,
               X_means = mean_val,
               X_sds = sd_val),
          file = file.path(outputdir, cruisename, "cmap-scaling-info.RDS"))

  ## Proportion of missing time points
  num_na <- apply(X_scaled, 2, function(mycol) sum(is.na(mycol))) 
  ## print(head(sort(num_na, decreasing = TRUE)))
  
  num_time_points <- nrow(X_scaled)
  prop_missing <- num_na / num_time_points
  
  ## Missingness distribution histogram
  png(
    file.path(figure_dir, "NA_distribution.png"),
    width = 8, height = 5, units = "in", res = 150
  )
    hist(prop_missing, breaks = 30, main = "Proportion of missing time points")
    abline(v = 0.25, col = "red", lty = 2)  
    missing_only_few <- sum(prop_missing < 0.25)
    legend("topleft", legend = paste0("Missing only\na few:\n", missing_only_few, "\nvariables"), box.lwd = 0)
    missing_a_lot <- sum(prop_missing > 0.25)
    legend("topright", legend = paste0("Missing \na lot:\n", missing_a_lot, "\nvariables"), box.lwd = 0)
  graphics.off()

  ## Visualize the feature matrix before imputation (variables with $<25\%$ missing values):
  X_scaled_copy0 <- X_scaled %>% as.matrix()
  dimnames(X_scaled_copy0) <- NULL
  
  ## Matrix Completion via Low-Rank SVD (`softImpute`); we'll
  ## fill missing values using nuclear-norm regularized singular value decomposition (`softImpute`).
  X_scaled_copy <- X_scaled_copy0[, prop_missing < 0.25] 

  if(sum(is.na(X_scaled_copy)) >= 1){

    ## Fit nuclear-norm regularized SVD model
    fit <- softImpute(X_scaled_copy, rank.max = 100, lambda = 10) ## I haven't carefully chosen lambda (or rank.max); we should do this..
    
    ## Complete matrix (fill NAs using low-rank fit)
    X_hat <- complete(X_scaled_copy, fit, unscale = FALSE)

    ## Compare imputed entries against original observed series:
    X_smaller <- X_scaled_copy
    observed_vec <- as.vector(X_smaller)
    p <- ncol(X_smaller)
    n <- nrow(X_smaller)
    
    df <- tibble(
      Time = rep(1:n, times = p),
      Series = rep(1:p, each = n),
      Value = as.vector(X_hat),
      Imputed_or_not = is.na(observed_vec)  
    )
    
    df_plot <- df %>%
      mutate(
        Series_Label = paste("Series", Series),
        Status = ifelse(Imputed_or_not, "Imputed", "Observed")
      )

    imputed_plot <- df_plot %>%
      group_by(Series) %>%
      filter(any(Imputed_or_not)) %>%
      ggplot(aes(x = Time, y = Value)) +
      geom_line(color = "gray50", linewidth = 0.5, alpha = 0.7) +
      geom_point(
        data = . %>% filter(Imputed_or_not),
        aes(color = Status),
        size = 0.3
      ) +
      facet_wrap(~ Series_Label, scales = "free_y") +
      scale_color_manual(
        values = c("Imputed" = "firebrick", "Observed" = "gray30")
      ) +
      labs(
        title = "Time Series Imputation via softImpute",
        ##| fig-cap: "Comparison of observed vs. softImpute filled values across series"
        x = "Time Index",
        y = "Value",
        color = "Data Point"
      ) +
      theme_minimal() +
      theme(
        legend.position = "bottom",
        strip.text = element_text(face = "bold")
      )

    ggsave(
      file.path(figure_dir, "imputation.png"),
      imputed_plot, width = 8, height = 5, units = "in"
    )

  } else {
    X_hat = X_scaled_copy0[, prop_missing < 0.25]
  }

  ## Sanity check: confirm non-imputed entries were untouched:
  reconstruction_errors <- (X_scaled_copy0[, prop_missing < 0.25] - X_hat) %>% 
    as.numeric() %>% 
    summary() %>%
    as.list() %>%
    as_tibble(check.names = FALSE)
  writeLines(
    as.character(
      make_html_table(
        reconstruction_errors,
        table_caption = "Summary of the reconstruction errors"
      )
    ),
    file.path(figure_dir, "reconstruction_errs.html")
  )

  ## Format the imputed output, re-attach spatial coordinates, and aggregate to hourly resolution.
  varnames = colnames(X_scaled)
  remaining_names = varnames[prop_missing < 0.25]
  colnames(X_hat) <- remaining_names
  
  X_imputed <- X_hat %>% 
    as_tibble() %>% 
    add_column(time = time0, lat = lat0, lon = lon0, .before = 1)
  
  X_imputed_hourly <- X_imputed %>% 
    group_by(time = lubridate::floor_date(time, "hour")) %>%
    summarize(across(where(is.numeric), mean), .groups = "drop")

  ## Save the imputed covariates in a single file;
  write.csv(X_imputed_hourly,
            file = file.path(outputdir, cruisename, "X-from-cmap-imputed-and-scaled.csv"),
            row.names = FALSE)

  ## Close the tab
  cat("\n\n")
}
